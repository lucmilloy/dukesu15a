async function checkAvailability() { const container = document.getElementById('results-container'); container.innerHTML = '

Loading latest ice availability...

'; try { // Fetch with cache-busting to always get the latest background update const response = await fetch('/ice_data.json?t=' + new Date().getTime()); const data = await response.json(); if (!data.slots || data.slots.length === 0) { container.innerHTML = `

No target 4:00 PM or weekend ice slots currently available in the system.

Last updated: ${data.last_updated || 'Just now'}

`; return; } let html = `

Last checked: ${data.last_updated}

`; data.slots.forEach(slot => { html += `
${slot.arena} - ${slot.date} @ ${slot.time}
Book on Ottawa.ca →
`; }); container.innerHTML = html; } catch (err) { container.innerHTML = '

Could not retrieve live slot data. Opening standard rinks...

'; } }
